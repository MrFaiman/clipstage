cmake_minimum_required(VERSION 3.28)
project(clipstage-obs VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)

add_library(clipstage-obs MODULE src/plugin.cpp)
target_link_libraries(clipstage-obs PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network)
set_target_properties(clipstage-obs PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(clipstage-obs PRIVATE /W4 /utf-8)
else()
  target_compile_options(clipstage-obs PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(APPLE)
  set_target_properties(clipstage-obs PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION "plugin" MACOSX_BUNDLE_GUI_IDENTIFIER "app.clipstage.obs")
endif()
install(TARGETS clipstage-obs LIBRARY DESTINATION obs-plugins BUNDLE DESTINATION obs-plugins RUNTIME DESTINATION obs-plugins/64bit)
